## 一、技术栈总览

### 1.1 前端技术栈

| 技术 | 版本 | 用途 | 选型理由 |
|------|------|------|----------|
| **React** | 18.x | UI 框架 | 生态成熟，pi-web-switch 可复用 |
| **TypeScript** | 5.x | 类型安全 | 提升代码质量，减少运行时错误 |
| **Vite** | 5.x | 构建工具 | 快速开发体验，HMR 即时生效 |
| **Tailwind CSS** | 3.x | 样式方案 | 原子化 CSS，快速开发，易于定制 |
| **Zustand** | 4.x | 状态管理 | 轻量级，API 简洁，pi-web-switch 已使用 |
| **React Router** | 6.x | 路由管理 | 声明式路由，嵌套路由支持 |
| **React Query** | 5.x | 数据获取 | 缓存管理，自动重试，乐观更新 |

### 1.2 后端技术栈

| 技术 | 版本 | 用途 | 选型理由 |
|------|------|------|----------|
| **Node.js** | 18.x+ | 运行时 | 与 PI 生态一致，异步 I/O 适合流式传输 |
| **Express** | 4.x | Web 框架 | 成熟稳定，中间件生态丰富 |
| **TypeScript** | 5.x | 类型安全 | 前后端统一类型定义 |
| **pnpm** | 8.x | 包管理器 | 速度快，磁盘占用小，monorepo 支持好 |
| **SQLite** | 3.x | 会话存储 | 轻量级，无需额外服务，单文件数据库 |
| **better-sqlite3** | 9.x | SQLite 驱动 | 同步 API，性能优秀，类型定义完整 |

### 1.3 通信协议

| 协议 | 用途 | 选型理由 |
|------|------|----------|
| **REST API** | 配置管理、会话管理 | 标准 HTTP 方法，易于理解和调试 |
| **SSE** | 对话流式传输 | 单向流式，实现简单，浏览器原生支持 |
| **WebSocket** | 备选方案 | 双向通信，但本项目 SSE 已足够 |

```typescript
  updateConfig(config: Partial<EngineConfig>): Promise<void>;

  listSessions(): Promise<SessionInfo[]>;
  
  /**
   * 获取会话详情

  getSession(id: string): Promise<SessionInfo>;

   * 删除会话

  deleteSession(id: string): Promise<void>;
  
  /**
   * 检查引擎是否可用
```

### 3.2 PI 适配器实现

```typescript
// packages/backend/src/adapters/PiAdapter.ts

import { EngineAdapter, ChatEvent, SessionContext } from '@greenhorn/shared';
import { createAgentSession, AgentSession } from '@pi-agent/coding-agent/sdk';

export class PiAdapter implements EngineAdapter {
  readonly name = 'PI';
  readonly version = '1.0.0';
  readonly description = 'PI 开源编码智能体';
  
  private session: AgentSession | null = null;
  
  async *sendMessage(
    prompt: string,
    context: SessionContext
  ): AsyncIterable<ChatEvent> {
    // 创建或复用会话
    if (!this.session || context.sessionId) {
      this.session = await createAgentSession({
        model: context.model || 'deepseek/deepseek-chat',
        cwd: process.cwd(),
        sessionId: context.sessionId,
      });
    }
    
    // 发送消息
    await this.session.prompt(prompt);
    
    // 监听事件并转换为统一格式
    for await (const event of this.session.events()) {
      yield this.mapToChatEvent(event);
    }
  }
  
  private mapToChatEvent(event: any): ChatEvent {
    switch (event.type) {
      case 'message_update':
        return { type: 'text', content: event.content };
      
      case 'thinking':
        return { type: 'thinking', content: event.content };
      
      case 'tool_execution_start':
        return {
          type: 'tool_call',
          name: event.toolName,
          input: event.input,
          id: event.id,
        };
      
      case 'tool_execution_end':
        return {
          type: 'tool_result',
          name: event.toolName,
          output: event.output,
          id: event.id,
        };
      
      case 'agent_end':
        return { type: 'done', reason: 'complete' };
      
      case 'error':
        return { type: 'error', message: event.message };
      
      default:
        return { type: 'text', content: '' };
    }
  }
  
  // ... 其他方法实现
}
```

### 3.3 轻量级对话引擎（方案B）

```typescript
// packages/backend/src/adapters/LightweightAdapter.ts

import { EngineAdapter, ChatEvent, SessionContext } from '@greenhorn/shared';

export class LightweightAdapter implements EngineAdapter {
  readonly name = 'Lightweight';
  readonly version = '1.0.0';
  readonly description = '轻量级对话引擎（直接调用 OpenAI 兼容 API）';
  
  async *sendMessage(
    prompt: string,
    context: SessionContext
  ): AsyncIterable<ChatEvent> {
    const response = await fetch(`${context.baseUrl}/v1/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${context.apiKey}`,
      },
      body: JSON.stringify({
        model: context.model,
        messages: [{ role: 'user', content: prompt }],
        stream: true,
        temperature: context.temperature,
        max_tokens: context.maxTokens,
      }),
    });
    
    if (!response.ok) {
      yield {
        type: 'error',
        message: `API 请求失败: ${response.statusText}`,
      };
      return;
    }
    
    // 解析 SSE 流
    for await (const chunk of this.parseSSEStream(response.body)) {
      yield chunk;
    }
    
    yield { type: 'done', reason: 'complete' };
  }
  
  private async *parseSSEStream(
    body: ReadableStream
  ): AsyncIterable<ChatEvent> {
    const reader = body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';
    
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      
      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';
      
      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = line.slice(6);
          if (data === '[DONE]') return;
          
          try {
            const parsed = JSON.parse(data);
            const content = parsed.choices[0]?.delta?.content;
            if (content) {
              yield { type: 'text', content };
            }
          } catch (e) {
            // 忽略解析错误
          }
        }
      }
    }
  }
  
  // ... 其他方法实现
}
```

### 3.4 SSE 流式端点

```typescript
// packages/backend/src/routes/chat.ts

import { Router } from 'express';
import { chatController } from '../controllers/chatController';

const router = Router();

/**
 * POST /api/chat
 * 发送消息并获取流式响应（SSE）
 */
router.post('/', async (req, res) => {
  const { prompt, context } = req.body;
  
  // 设置 SSE 响应头
  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');
  
  try {
    const events = await chatController.sendMessage(prompt, context);
    
    for await (const event of events) {
      res.write(`data: ${JSON.stringify(event)}\n\n`);
    }
    
    res.end();
  } catch (error) {
    res.write(`data: ${JSON.stringify({
      type: 'error',
      message: error.message,
    })}\n\n`);
    res.end();
  }
});

export default router;
```

### 3.5 前端 SSE Hook

```typescript
// packages/frontend/src/hooks/useSSE.ts

import { useState, useCallback } from 'react';
import { ChatEvent } from '@greenhorn/shared';

export function useSSE() {
  const [events, setEvents] = useState<ChatEvent[]>([]);
  const [isStreaming, setIsStreaming] = useState(false);
  
  const sendMessage = useCallback(async (
    prompt: string,
    context: any
  ) => {
    setIsStreaming(true);
    setEvents([]);
    
    const response = await fetch('/api/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt, context }),
    });
    
    const reader = response.body?.getReader();
    const decoder = new TextDecoder();
    
    if (!reader) return;
    
    let buffer = '';
    
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      
      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n\n');
      buffer = lines.pop() || '';
      
      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = JSON.parse(line.slice(6));
          setEvents(prev => [...prev, data]);
        }
      }
    }
    
    setIsStreaming(false);
  }, []);
  
  return { events, isStreaming, sendMessage };
}
```
